\section{Phonetic Clustering and Permutation Testing}
This appendix provides the Python implementation of the unguided clustering and significance testing utilized in Chapter 4. 

\begin{verbatim}
import numpy as np
from sklearn.decomposition import PCA

def levenshtein_dist(s1, s2):
    """Calculates the normalized Levenshtein distance."""
    # Normalized by max length to ensure distance is in range [0, 1]
    if len(s1) < len(s2): return levenshtein_dist(s2, s1)
    if len(s2) == 0: return 1.0
    previous_row = range(len(s2) + 1)
    for i, c1 in enumerate(s1):
        current_row = [i + 1]
        for j, c2 in enumerate(s2):
            insertions = previous_row[j + 1] + 1
            deletions = current_row[j] + 1
            substitutions = previous_row[j] + (c1 != c2)
            current_row.append(min(insertions, deletions, substitutions))
        previous_row = current_row
    return previous_row[-1] / max(len(s1), len(s2))

\section{Data Pipeline and Normalization}
The Mahadevan/Wells Harappan corpus was normalized using the following protocols:
\begin{enumerate}
    \item \textbf{Tokenization:} Sequences were extracted from the M-code concordance (+XXX-YYY+ format).
    \item \textbf{Purging:} Delimiter artifacts ('000', '0') and fragmentary inscriptions (length < 3) were strictly removed.
    \item \textbf{Smoothing:} Laplacian additive smoothing ($\epsilon = 10^{-6}$) was applied to handle out-of-vocabulary transitions during cross-validation.
\end{enumerate}
